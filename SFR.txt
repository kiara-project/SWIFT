# Star Formation History Logger file
######################################################
# The quantities are all given in internal physical units!
#
# (0)  Simulation step
#      Unit = dimensionless
# (1) Time since Big Bang (cosmological run), Time since start of the simulation (non-cosmological run).
#     Unit = 3.085678e+19 s
#     Unit = 9.778131e+11 yr
#     Unit = 9.778131e+05 Myr
# (2) Scale factor
#     Unit = dimensionless
# (3) Redshift
#     Unit = dimensionless
# (4) Total mass stars formed in the current time-step.
#     Unit = 1.988410e+43 gram
#     Unit = 1.000000e+10 Msun
# (5) The total SFR of all the active particles.
#     Unit = 6.443998e+23 gram/s
#     Unit = 1.022690e-02 Msun/yr
# (6) The star formation rate (SFR) of active particles multiplied by their time-step size.
#     Unit = 1.988410e+43 gram
#     Unit = 1.000000e+10 Msun
# (7) The total SFR of all the particles in the simulation.
#     Unit = 6.443998e+23 gram/s
#     Unit = 1.022690e-02 Msun/yr
#
# (0)         (1)            (2)          (3)            (4)            (5)            (6)            (7)
#            Time             a            z        total M_stars  SFR (active)  SFR*dt (active)  SFR (total)
